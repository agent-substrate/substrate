The file is named by its path from the repository root, taken from the "file" argument or, failing that, the page's repo_source front matter. It must be mounted under assets/repo in hugo.yaml. Its front matter is not rendered (page-params.html reads it) and the page title replaces its leading H1. Relative links are rewritten by rewrite-repo-links.html. Callers must use {{% %}} so that Hugo renders the returned Markdown. */ -}} {{- $file := .Get "file" | default (.Page.Params.repo_source) -}} {{- if not $file -}} {{- errorf "include-file: %s: no file argument and no repo_source front matter" .Page.File.Path -}} {{- end -}} {{- $doc := partialCached "repo-doc.html" $file $file -}} {{- if not $doc.found -}} {{- errorf "include-file: %s: %q is not mounted under assets/repo; add it to module.mounts in hugo.yaml" .Page.File.Path $file -}} {{- end -}} {{- $content := $doc.body -}} {{- if .Get "shift_headings" -}} {{- $content = replaceRE `(?m)^(#{2,5}) ` "#$1 " $content -}} {{- end -}} {{- $content = partial "rewrite-repo-links.html" (dict "content" $content "dir" (path.Dir $file)) -}} {{/* Emit Markdown for the {{% %}} caller to render; rendering it here as well would run Goldmark twice. safeHTML keeps raw HTML in the docs intact. */ -}} {{- $content | safeHTML -}}
