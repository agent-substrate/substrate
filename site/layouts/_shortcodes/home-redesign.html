GET STARTED

Your first actor in five minutes

# 1 · create a cluster and install the system
hack/create-kind-cluster.sh
hack/install-ate-kind.sh --deploy-ate-system --deploy-demo-counter

# 2 · create an actor
go install ./cmd/kubectl-ate
kubectl ate create actor my-counter-1 -a ate-demo-counter --template counter

# 3 · send it a request (it resumes on demand)
curl -X POST -H "ate-target-actor: ate-demo-counter/my-counter-1" \
  http://localhost:8000/
